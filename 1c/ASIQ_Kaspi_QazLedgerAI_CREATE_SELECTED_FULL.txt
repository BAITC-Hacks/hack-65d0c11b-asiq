// АСИК Kaspi 1.2.0 SAFE-DRAFT + QazLedger AI TABLE.
// Форма импортирует, сверяет и по отдельной команде создает только
// непроведенные черновики только отмеченных пользователем операций.
// QazLedger AI анализирует все группы загруженной выписки и
// выводит результат в реквизит формы "РезультатыAI".
// AI ничего не создаёт и не проводит в 1С.

&НаКлиенте
Процедура Импортировать(Команда)

	АдресВременногоХранилища = "";
	ИсходноеИмяФайла = "";
	ВыбранноеИмяФайла = "";

	Если Не ПоместитьФайл(АдресВременногоХранилища, ИсходноеИмяФайла, ВыбранноеИмяФайла, Истина, УникальныйИдентификатор) Тогда
		Возврат;
	КонецЕсли;

	Если НРег(Прав(ВыбранноеИмяФайла, 5)) <> ".xlsx" Тогда
		Предупреждение("Выберите выписку Kaspi в формате XLSX.");
		Возврат;
	КонецЕсли;

	Попытка
		ИмпортироватьНаСервере(АдресВременногоХранилища);
	Исключение
		Предупреждение("Не удалось прочитать выписку: " + КраткоеПредставлениеОшибки(ИнформацияОбОшибке()));
		Возврат;
	КонецПопытки;

	ИмяФайла = ВыбранноеИмяФайла;

	Сообщить("Выписка прочитана. Документы не создавались и не изменялись.");

КонецПроцедуры

&НаСервере
Процедура ИмпортироватьНаСервере(АдресВременногоХранилища)

	Результат = АСИК_ИмпортВыпискиKaspiСервер.ПрочитатьИзВременногоХранилища(АдресВременногоХранилища);
	ЗначениеВРеквизитФормы(Результат.Строки, "СтрокиВыписки");
	ЗначениеВРеквизитФормы(Результат.Группы, "Группы");

	ПустыеРезультатыAI = СоздатьТаблицуРезультатовAI();
	ЗначениеВРеквизитФормы(ПустыеРезультатыAI, "РезультатыAI");

	Итоги = Результат.Итоги;
	АдресДанныхИмпорта = ПоместитьВоВременноеХранилище(
		Результат.Группы, УникальныйИдентификатор);

&НаКлиенте
Процедура СоздатьЧерновики(Команда)

	Если Группы.Количество() = 0 Тогда
		Предупреждение("Сначала загрузите выписку Kaspi.");
		Возврат;
	КонецЕсли;

	Если РезультатыAI.Количество() = 0 Тогда
		Предупреждение("Сначала выполните «Анализировать с AI».");
		Возврат;
	КонецЕсли;

	КоличествоВыбранных = 0;

	Для Каждого СтрокаAI Из РезультатыAI Цикл
		Если СтрокаAI.КСозданию Тогда
			КоличествоВыбранных = КоличествоВыбранных + 1;
		КонецЕсли;
	КонецЦикла;

	Если КоличествоВыбранных = 0 Тогда
		Предупреждение("Не отмечено ни одной строки «К созданию».");
		Возврат;
	КонецЕсли;

	ТекстВопроса =
		"Будут созданы только НЕПРОВЕДЕННЫЕ черновики отмеченных строк: "
		+ Строка(КоличествоВыбранных)
		+ ". Продолжить?";

	Если Вопрос(ТекстВопроса, РежимДиалогаВопрос.ДаНет, 0)
		<> КодВозвратаДиалога.Да Тогда
		Возврат;
	КонецЕсли;

	Попытка
		РезультатСоздания = СоздатьЧерновикиНаСервере();
	Исключение
		Предупреждение("Черновики не созданы: "
			+ КраткоеПредставлениеОшибки(ИнформацияОбОшибке()));
		Возврат;
	КонецПопытки;

	Сообщить(РезультатСоздания.Текст);

&НаСервере
Функция СоздатьЧерновикиНаСервере()

	Если ПустаяСтрока(АдресДанныхИмпорта) Тогда
		ВызватьИсключение "Данные выписки не найдены. Загрузите XLSX повторно.";
	КонецЕсли;

	ТаблицаГрупп = ПолучитьИзВременногоХранилища(АдресДанныхИмпорта);

	Если ТаблицаГрупп = Неопределено Тогда
		ВызватьИсключение "Данные выписки больше не доступны. Загрузите XLSX повторно.";
	КонецЕсли;

	ТаблицаAI = РеквизитФормыВЗначение("РезультатыAI");

	Если ТаблицаAI = Неопределено Или ТаблицаAI.Количество() = 0 Тогда
		ВызватьИсключение "Нет результатов AI для создания.";
	КонецЕсли;

	Если ТаблицаAI.Количество() <> ТаблицаГрупп.Количество() Тогда
		ВызватьИсключение
			"Состав выписки изменился после AI-анализа. "
			+ "Выполните «Анализировать с AI» повторно.";
	КонецЕсли;

	ВыбранныеГруппы = ТаблицаГрупп.СкопироватьКолонки();

	Для Индекс = 0 По ТаблицаAI.Количество() - 1 Цикл

		Если ТаблицаAI[Индекс].КСозданию Тогда

			НоваяГруппа = ВыбранныеГруппы.Добавить();
			ЗаполнитьЗначенияСвойств(
				НоваяГруппа,
				ТаблицаГрупп[Индекс]);

		КонецЕсли;

	КонецЦикла;

	Если ВыбранныеГруппы.Количество() = 0 Тогда
		ВызватьИсключение "Не отмечено ни одной строки «К созданию».";
	КонецЕсли;

	// Используем уже существующую безопасную логику расширения:
	// документы только записываются как НЕПРОВЕДЕННЫЕ.
	РезультатСоздания =
		АСИК_ИмпортВыпискиKaspiСервер.СоздатьБезопасныеЧерновикиКомиссий(
			ВыбранныеГруппы);

	// После создания обновляем сверку полного набора операций.
	ТаблицаГрупп =
		АСИК_ИмпортВыпискиKaspiСервер.СверитьГруппыСБанковскимиДокументами(
			ТаблицаГрупп);

	АдресДанныхИмпорта = ПоместитьВоВременноеХранилище(
		ТаблицаГрупп,
		УникальныйИдентификатор);

	ЗначениеВРеквизитФормы(
		ТаблицаГрупп,
		"Группы");

	Возврат РезультатСоздания;

&НаКлиенте
Процедура АнализироватьAI(Команда)

	Если Группы.Количество() = 0 Тогда
		Предупреждение("Сначала загрузите выписку Kaspi.");
		Возврат;
	КонецЕсли;

			+ ", ручная проверка "

			+ Строка(Результат.Ошибки)
			+ ".");

	Исключение

			"Ошибка QazLedger AI: "

				ИнформацияОбОшибке()));

	КонецПопытки;

&НаСервере
Функция АнализироватьAIНаСервере()

	Если ПустаяСтрока(АдресДанныхИмпорта) Тогда
		ВызватьИсключение "Сначала загрузите выписку Kaspi.";
	КонецЕсли;

	ТаблицаГрупп = ПолучитьИзВременногоХранилища(АдресДанныхИмпорта);

	Если ТаблицаГрупп = Неопределено Тогда
		ВызватьИсключение "Данные выписки больше не доступны. Загрузите XLSX повторно.";
	КонецЕсли;

	Если ТаблицаГрупп.Количество() = 0 Тогда
		ВызватьИсключение "В выписке нет операций для анализа.";
	КонецЕсли;

	ТаблицаРезультатов = СоздатьТаблицуРезультатовAI();

	Всего = ТаблицаГрупп.Количество();
	Успешно = 0;
	РучнаяПроверка = 0;
	Ошибки = 0;

	Для Каждого СтрокаГруппы Из ТаблицаГрупп Цикл

		СтрокаРезультата = ТаблицаРезультатов.Добавить();
		СтрокаРезультата.КСозданию = Ложь;
		СтрокаРезультата.Дата = СтрокаГруппы.Дата;
		СтрокаРезультата.Сумма = СтрокаГруппы.Сумма;
		СтрокаРезультата.Контрагент = Строка(СтрокаГруппы.Корреспондент);

		Попытка

			ТекстОтвета = ОтправитьГруппуВQazLedgerAI(СтрокаГруппы);
			Анализ = РазобратьОтветQazLedgerAI(ТекстОтвета);

			Успешно = Успешно + 1;

			СтрокаРезультата.ПравилоAI = Анализ.Rule;
			СтрокаРезультата.РешениеAI = Анализ.OperationName;
			СтрокаРезультата.Уверенность = Окр(Анализ.Confidence * 100, 0);
			СтрокаРезультата.Причина = Анализ.Reason;

			Если Анализ.ManualReview Тогда
				РучнаяПроверка = РучнаяПроверка + 1;
				СтрокаРезультата.Статус = "РУЧНАЯ ПРОВЕРКА";
			Иначе
				СтрокаРезультата.Статус = "OK";
			КонецЕсли;

		Исключение

			Ошибки = Ошибки + 1;
			СтрокаРезультата.ПравилоAI = "";
			СтрокаРезультата.РешениеAI = "Ошибка AI";
			СтрокаРезультата.Уверенность = 0;
			СтрокаРезультата.Статус = "ОШИБКА";
			СтрокаРезультата.Причина =
				КраткоеПредставлениеОшибки(
					ИнформацияОбОшибке());

		КонецПопытки;

	КонецЦикла;

	ЗначениеВРеквизитФормы(
		ТаблицаРезультатов,
		"РезультатыAI");

	Возврат Новый Структура(
		"Всего,Успешно,РучнаяПроверка,Ошибки",
		Всего,
		Успешно,
		РучнаяПроверка,
		Ошибки);

&НаСервереБезКонтекста
Функция ОтправитьГруппуВQazLedgerAI(СтрокаГруппы)

	НаправлениеОперации = "unknown";

	Если СтрокаГруппы.Направление = "Поступление" Тогда
		НаправлениеОперации = "incoming";
	ИначеЕсли СтрокаГруппы.Направление = "Списание" Тогда
		НаправлениеОперации = "outgoing";
	КонецЕсли;

	Данные = Новый Структура;
	Данные.Вставить("bank", "Kaspi Bank");
	Данные.Вставить("date", Формат(СтрокаГруппы.Дата, "ДФ=yyyy-MM-dd"));
	Данные.Вставить("amount", СтрокаГруппы.Сумма);
	Данные.Вставить("direction", НаправлениеОперации);
	Данные.Вставить("counterparty", Строка(СтрокаГруппы.Корреспондент));
	Данные.Вставить("bin", Строка(СтрокаГруппы.БИНКорреспондента));
	Данные.Вставить("iban", Строка(СтрокаГруппы.ИИККорреспондента));
	Данные.Вставить("bik", Строка(СтрокаГруппы.БИККорреспондента));
	Данные.Вставить("knp", Строка(СтрокаГруппы.КНП));
	Данные.Вставить("purpose", Строка(СтрокаГруппы.НазначениеПлатежа));

	ЗаписьJSON = Новый ЗаписьJSON;
	ЗаписьJSON.УстановитьСтроку();
	ЗаписатьJSON(ЗаписьJSON, Данные);
	ТелоJSON = ЗаписьJSON.Закрыть();

	ЗащищенноеСоединение = Новый ЗащищенноеСоединениеOpenSSL;

	HTTP = Новый HTTPСоединение(
		"qazledger-ai-deploy-production.up.railway.app",
		443,
		,
		,
		,
		30,
		ЗащищенноеСоединение);

	Запрос = Новый HTTPЗапрос("/api/analyze");
	Запрос.Заголовки.Вставить(
		"Content-Type",
		"application/json; charset=utf-8");

	Запрос.УстановитьТелоИзСтроки(
		ТелоJSON,
		КодировкаТекста.UTF8);

	Ответ = HTTP.ОтправитьДляОбработки(Запрос);
	ТекстОтвета = Ответ.ПолучитьТелоКакСтроку();

	Если Ответ.КодСостояния < 200
		Или Ответ.КодСостояния >= 300 Тогда

		ВызватьИсключение
			"HTTP "
			+ Строка(Ответ.КодСостояния)
			+ ": "
			+ ТекстОтвета;

	КонецЕсли;

	Возврат ТекстОтвета;

&НаСервереБезКонтекста
Функция РазобратьОтветQazLedgerAI(ТекстОтвета)

	Чтение = Новый ЧтениеJSON;
	Чтение.УстановитьСтроку(ТекстОтвета);

	ДанныеОтвета = ПрочитатьJSON(Чтение);
	Чтение.Закрыть();

	Если ДанныеОтвета = Неопределено Тогда
		ВызватьИсключение "Пустой JSON-ответ.";
	КонецЕсли;

	Успех = Ложь;

	Если Не ДанныеОтвета.Свойство("success", Успех) Тогда
		ВызватьИсключение "В ответе AI отсутствует поле success.";
	КонецЕсли;

	Если Не Успех Тогда
		ВызватьИсключение "AI вернул success=false.";
	КонецЕсли;

	АнализJSON = Неопределено;

	Если Не ДанныеОтвета.Свойство("analysis", АнализJSON) Тогда
		ВызватьИсключение "В ответе AI отсутствует блок analysis.";
	КонецЕсли;

	Rule = "";
	OperationName = "";
	Confidence = 0;
	ManualReview = Истина;
	Reason = "";

	АнализJSON.Свойство("rule", Rule);
	АнализJSON.Свойство("operation_name", OperationName);
	АнализJSON.Свойство("confidence", Confidence);
	АнализJSON.Свойство("manual_review", ManualReview);
	АнализJSON.Свойство("reason", Reason);

	Возврат Новый Структура(
		"Rule,OperationName,Confidence,ManualReview,Reason",
		Rule,
		OperationName,
		Confidence,
		ManualReview,
		Reason);
