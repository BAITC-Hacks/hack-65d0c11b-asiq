	Если ПустаяСтрока(АдресВременногоХранилища) Тогда

Функция ПрочитатьФайлKaspi(ИмяФайла) Экспорт

	ТабличныйДокумент = Новый ТабличныйДокумент;
	ТабличныйДокумент.Прочитать(ИмяФайла);
	СчетКлиента = НайтиСчетКлиентаKaspi(ТабличныйДокумент);

	СтрокаЗаголовка = НайтиСтрокуЗаголовка(ТабличныйДокумент);
	Если СтрокаЗаголовка = 0 Тогда
		ВызватьИсключение "Не найдены колонки Kaspi: № документа, Дата операции, Дебет и Кредит.";
	КонецЕсли;

	Строки = СоздатьТаблицуСтрок();
	ПустыхСтрокПодряд = 0;
	ПерваяСтрокаДанных = СтрокаЗаголовка + 1;
	Если ТекстЯчейки(ТабличныйДокумент, ПерваяСтрокаДанных, 1) = "1"
		И ТекстЯчейки(ТабличныйДокумент, ПерваяСтрокаДанных, 2) = "2" Тогда
		ПерваяСтрокаДанных = ПерваяСтрокаДанных + 1;
	КонецЕсли;

	Для НомерСтроки = ПерваяСтрокаДанных По ТабличныйДокумент.ВысотаТаблицы Цикл

		НомерДокумента = ТекстЯчейки(ТабличныйДокумент, НомерСтроки, 1);
		ТекстДаты = ТекстЯчейки(ТабличныйДокумент, НомерСтроки, 2);

		Если ПустаяСтрока(НомерДокумента) И ПустаяСтрока(ТекстДаты) Тогда
			ПустыхСтрокПодряд = ПустыхСтрокПодряд + 1;
			Если ПустыхСтрокПодряд >= 5 Тогда
				Прервать;
			КонецЕсли;
			Продолжить;
		КонецЕсли;

		ПустыхСтрокПодряд = 0;

		ДатаОперации = ПрочитатьДатуKaspi(ТекстДаты);
		Если ДатаОперации = Неопределено Тогда
			Продолжить;
		КонецЕсли;

		НоваяСтрока = Строки.Добавить();
		НоваяСтрока.НомерСтроки = НомерСтроки;
		НоваяСтрока.СчетКлиента = СчетКлиента;
		НоваяСтрока.НомерДокументаБанка = НомерДокумента;
		НоваяСтрока.ДатаОперации = ДатаОперации;
		НоваяСтрока.Дебет = ПрочитатьЧислоKaspi(ТекстЯчейки(ТабличныйДокумент, НомерСтроки, 3));
		НоваяСтрока.Кредит = ПрочитатьЧислоKaspi(ТекстЯчейки(ТабличныйДокумент, НомерСтроки, 4));
		НоваяСтрока.Корреспондент = ТекстЯчейки(ТабличныйДокумент, НомерСтроки, 5);
		НоваяСтрока.ИИК = ТекстЯчейки(ТабличныйДокумент, НомерСтроки, 6);
		НоваяСтрока.БИК = ТекстЯчейки(ТабличныйДокумент, НомерСтроки, 7);
		НоваяСтрока.КНП = ТекстЯчейки(ТабличныйДокумент, НомерСтроки, 8);
		НоваяСтрока.НазначениеПлатежа = ТекстЯчейки(ТабличныйДокумент, НомерСтроки, 9);
		НоваяСтрока.БИНКорреспондента = ИзвлечьБИН(НоваяСтрока.Корреспондент);
		НоваяСтрока.Направление = НаправлениеСтроки(НоваяСтрока.Дебет, НоваяСтрока.Кредит);
		НоваяСтрока.Правило = ОпределитьПравило(НоваяСтрока);
		НоваяСтрока.КлючГруппы = КлючГруппы(НоваяСтрока);

	КонецЦикла;

	Если Строки.Количество() = 0 Тогда
		ВызватьИсключение "В выписке не найдено ни одной операции.";
	КонецЕсли;

	Группы = СгруппироватьСтроки(Строки);
	Итоги = СформироватьИтоги(Строки, Группы);

	Возврат Новый Структура("Строки,Группы,Итоги", Строки, Группы, Итоги);

Функция НайтиСтрокуЗаголовка(ТабличныйДокумент)

	ПоследняяСтрока = ТабличныйДокумент.ВысотаТаблицы;
	Если ПоследняяСтрока > 40 Тогда
		ПоследняяСтрока = 40;
	КонецЕсли;

	Для НомерСтроки = 1 По ПоследняяСтрока Цикл
		Колонка1 = НРег(НормализоватьТекст(ТекстЯчейки(ТабличныйДокумент, НомерСтроки, 1)));
		Колонка2 = НРег(НормализоватьТекст(ТекстЯчейки(ТабличныйДокумент, НомерСтроки, 2)));
		Колонка3 = НРег(НормализоватьТекст(ТекстЯчейки(ТабличныйДокумент, НомерСтроки, 3)));
		Колонка4 = НРег(НормализоватьТекст(ТекстЯчейки(ТабличныйДокумент, НомерСтроки, 4)));

		Если СтрНайти(Колонка1, "документ") > 0
			И СтрНайти(Колонка2, "дата") > 0
			И Колонка3 = "дебет"
			И Колонка4 = "кредит" Тогда
			Возврат НомерСтроки;
		КонецЕсли;
	КонецЦикла;

	Возврат 0;

Функция НайтиСчетКлиентаKaspi(ТабличныйДокумент)

	ПоследняяСтрока = ТабличныйДокумент.ВысотаТаблицы;
	Если ПоследняяСтрока > 20 Тогда
		ПоследняяСтрока = 20;
	КонецЕсли;

	ПоследняяКолонка = ТабличныйДокумент.ШиринаТаблицы;
	Если ПоследняяКолонка > 5 Тогда
		ПоследняяКолонка = 5;
	КонецЕсли;

	Для НомерСтроки = 1 По ПоследняяСтрока Цикл
		Для НомерКолонки = 1 По ПоследняяКолонка Цикл
			Текст = ВРег(НормализоватьТекст(
				ТекстЯчейки(ТабличныйДокумент, НомерСтроки, НомерКолонки)));
			ПозицияKZ = СтрНайти(Текст, "KZ");
			Если ПозицияKZ = 0 Тогда
				Продолжить;
			КонецЕсли;

			Кандидат = Сред(Текст, ПозицияKZ, 20);
			Кандидат = СтрЗаменить(Кандидат, " ", "");
			Если СтрДлина(Кандидат) = 20 И Лев(Кандидат, 2) = "KZ" Тогда
				Возврат Кандидат;
			КонецЕсли;
		КонецЦикла;
	КонецЦикла;

	Возврат "";

Функция СгруппироватьСтроки(Строки)

	Группы = СоздатьТаблицуГрупп();
	ИндексГрупп = Новый Соответствие;

	Для Каждого СтрокаВыписки Из Строки Цикл
		СтрокаГруппы = ИндексГрупп.Получить(СтрокаВыписки.КлючГруппы);

		Если СтрокаГруппы = Неопределено Тогда
			СтрокаГруппы = Группы.Добавить();
			СтрокаГруппы.КлючГруппы = СтрокаВыписки.КлючГруппы;
			СтрокаГруппы.Дата = НачалоДня(СтрокаВыписки.ДатаОперации);
			СтрокаГруппы.Направление = СтрокаВыписки.Направление;
			СтрокаГруппы.Сумма = 0;
			СтрокаГруппы.КоличествоСтрок = 0;
			СтрокаГруппы.Корреспондент = СтрокаВыписки.Корреспондент;
			СтрокаГруппы.БИНКорреспондента = СтрокаВыписки.БИНКорреспондента;
			СтрокаГруппы.КНП = СтрокаВыписки.КНП;
			СтрокаГруппы.Правило = СтрокаВыписки.Правило;
			Рекомендация = РекомендацияПоПравилу(СтрокаВыписки.Правило);
			СтрокаГруппы.ПредлагаемыйДокумент = Рекомендация.Документ;
			СтрокаГруппы.Комментарий = Рекомендация.Комментарий;
			СтрокаГруппы.СтатусСверки = "Не проверено";
			СтрокаГруппы.КоличествоСовпадений = 0;
			СтрокаГруппы.НайденныеДокументы = "";
			СтрокаГруппы.НомерДокументаБанка = СтрокаВыписки.НомерДокументаБанка;
			СтрокаГруппы.НазначениеПлатежа = СтрокаВыписки.НазначениеПлатежа;
			СтрокаГруппы.ИИККорреспондента = СтрокаВыписки.ИИК;
			СтрокаГруппы.БИККорреспондента = СтрокаВыписки.БИК;
			СтрокаГруппы.СчетКлиента = СтрокаВыписки.СчетКлиента;
			СтрокаГруппы.РазрешеноСоздание =
				ЭтоПоддерживаемоеПравилоKaspi(СтрокаВыписки.Правило);
			ИндексГрупп.Вставить(СтрокаВыписки.КлючГруппы, СтрокаГруппы);
		КонецЕсли;

		Если СтрокаВыписки.Направление = "Поступление" Тогда
			СтрокаГруппы.Сумма = СтрокаГруппы.Сумма + СтрокаВыписки.Кредит;
		Иначе
			СтрокаГруппы.Сумма = СтрокаГруппы.Сумма + СтрокаВыписки.Дебет;
		КонецЕсли;
		СтрокаГруппы.КоличествоСтрок = СтрокаГруппы.КоличествоСтрок + 1;
		Если СтрокаГруппы.КоличествоСтрок > 1 Тогда
			// У сводной дневной комиссии нет одного достоверного номера банка.
			СтрокаГруппы.НомерДокументаБанка = "";
		КонецЕсли;
	КонецЦикла;

	Возврат Группы;

Функция ОпределитьПравило(СтрокаВыписки)

	Назначение = НРег(НормализоватьТекст(СтрокаВыписки.НазначениеПлатежа));
	Корреспондент = НРег(НормализоватьТекст(СтрокаВыписки.Корреспондент));
	КНП = СокрЛП(СтрокаВыписки.КНП);

	Если СтрокаВыписки.Кредит > 0 И КНП = "190"
		И СтрНайти(Назначение, "продажи с kaspi.kz") = 1 Тогда
		Возврат "SALE";
	КонецЕсли;

	Если СтрокаВыписки.Кредит > 0 И КНП = "343"
		И СтрНайти(Назначение, "со своего текущего счета") > 0
		И СтрНайти(Назначение, "на свой текущий счет") > 0 Тогда
		Возврат "OWN_TRANSFER_IN";
	КонецЕсли;

	Если СтрокаВыписки.Дебет > 0 И КНП = "343"
		И СтрНайти(Назначение, "собствен") > 0
		И СтрНайти(Назначение, "счет") > 0 Тогда
		Возврат "OWN_TRANSFER_OUT";
	КонецЕсли;

	Если СтрокаВыписки.Дебет > 0 И КНП = "190"
		И СтрНайти(Назначение, "возврат продаж") = 1 Тогда
		Возврат "SALE_RETURN_REVIEW";
	КонецЕсли;

	Если СтрокаВыписки.Дебет > 0 И КНП = "710"
		И (СтрокаВыписки.БИНКорреспондента = "860108450870"
			Или СтрНайти(Корреспондент, "абакумова аурика") > 0) Тогда
		Возврат "ACCOUNTABLE_AURIKA";
	КонецЕсли;

	Если СтрокаВыписки.Дебет > 0 И КНП = "841"
		И (СтрНайти(Корреспондент, "kaspi bank") > 0 Или СтрокаВыписки.БИНКорреспондента = "971240001315") Тогда
		Возврат "KASPI_BANK_FEE";
	КонецЕсли;

	Если СтрокаВыписки.Дебет > 0 И КНП = "851"
		И (СтрНайти(Корреспондент, "kaspi pay") > 0 Или СтрокаВыписки.БИНКорреспондента = "200840000951") Тогда
		Возврат "KASPI_PAY_FEE";
	КонецЕсли;

	Если СтрокаВыписки.Дебет > 0 И КНП = "851"
		И (СтрНайти(Корреспондент, "kaspi магазин") > 0 Или СтрокаВыписки.БИНКорреспондента = "150540002688") Тогда
		Возврат "KASPI_SHOP_FEE";
	КонецЕсли;

	Если СтрокаВыписки.Кредит > 0 И КНП = "880" Тогда
		Возврат "KASPI_FEE_REFUND";
	КонецЕсли;

	Возврат "MANUAL_REVIEW";

Функция КлючГруппы(СтрокаВыписки)

	ДатаКлюча = Формат(НачалоДня(СтрокаВыписки.ДатаОперации), "ДФ=yyyyMMdd");

	Если СтрокаВыписки.Правило = "SALE" Тогда
		Возврат СтрокаВыписки.Правило + "|" + ДатаКлюча;
	КонецЕсли;

	Если СтрокаВыписки.Правило = "KASPI_BANK_FEE"

Функция НайтиПроведенныеИсходящиеПереводыKaspi(
		ДатаДокумента, СуммаДокумента,

		Новый Массив, Новый Массив, Истина, "", Неопределено);

	НомерСчетаKaspi = НормализоватьНомерСчетаKaspi(НомерСчетаKaspi);
	НомерСчетаДругогоБанка =
		НормализоватьНомерСчетаKaspi(НомерСчетаДругогоБанка);
	Если ПустаяСтрока(НомерСчетаKaspi)
		Или ПустаяСтрока(НомерСчетаДругогоБанка) Тогда
		ДобавитьПричинуНеполнойПроверкиKaspi(
			Результат, "не получены оба ИИК внутреннего перевода");
		Возврат Результат;
	КонецЕсли;
	Если НомерСчетаKaspi = НомерСчетаДругогоБанка Тогда
		ДобавитьПричинуНеполнойПроверкиKaspi(
			Результат, "ИИК отправителя и получателя совпадают");
		Возврат Результат;
	КонецЕсли;

	СчетKaspi = НайтиБанковскийСчетОрганизацииKaspi(НомерСчетаKaspi);
	СчетДругогоБанка = НайтиБанковскийСчетОрганизацииKaspi(
		НомерСчетаДругогоБанка);
	Если СчетKaspi = Неопределено
		Или СчетДругогоБанка = Неопределено Тогда
		ДобавитьПричинуНеполнойПроверкиKaspi(
			Результат,
			"один или оба банковских счета не найдены однозначно в справочнике");

	КонецЕсли;

	ОрганизацияДругогоБанка =
		ПолучитьОрганизациюБанковскогоСчетаKaspi(СчетДругогоБанка);
	Если ОрганизацияKaspi = Неопределено
		Или ОрганизацияДругогоБанка = Неопределено Тогда
		ДобавитьПричинуНеполнойПроверкиKaspi(
			Результат, "не удалось определить владельца одного из счетов");
		Возврат Результат;
	КонецЕсли;
	Если ОрганизацияKaspi <> ОрганизацияДругогоБанка Тогда
		ДобавитьПричинуНеполнойПроверкиKaspi(
			Результат, "банковские счета принадлежат разным владельцам");
		Возврат Результат;
	КонецЕсли;
	Результат.Организация = ОрганизацияKaspi;

	ИменаДокументов = Новый Массив;
	ИменаДокументов.Добавить("ПлатежноеПоручениеИсходящее");
	ИменаДокументов.Добавить("ПлатежныйОрдерСписаниеДенежныхСредств");
	ТипДокументаНайден = Ложь;

	Для Каждого ИмяДокумента Из ИменаДокументов Цикл
		Если Метаданные.Документы.Найти(ИмяДокумента) = Неопределено Тогда
			Продолжить;
		КонецЕсли;
		ТипДокументаНайден = Истина;
		ДобавитьКандидатыВстречногоПереводаKaspi(
			Результат, ИмяДокумента,
			ДатаДокумента, СуммаДокумента,
			СчетKaspi, НомерСчетаKaspi,
			СчетДругогоБанка, НомерСчетаДругогоБанка);
	КонецЦикла;

	Если Не ТипДокументаНайден Тогда
		ДобавитьПричинуНеполнойПроверкиKaspi(
			Результат, "в конфигурации нет поддерживаемого исходящего документа");
	КонецЕсли;

Функция СоздатьБезопасныеЧерновикиКомиссий(Группы) Экспорт

	Создано = 0;
	ПропущеноДублей = 0;
	УчтеноДругимБанком = 0;
	ПропущеноБезОбразца = 0;
	Ошибок = 0;
	ПерваяСсылка = Неопределено;
	Сообщения = Новый Массив;

	Для Каждого Группа Из Группы Цикл

		Если ЭтоБезопаснаяКомиссияKaspi(Группа.Правило) Тогда
			РезультатСоздания = СоздатьЧерновикКомиссииKaspi(Группа);
		Иначе
			РезультатСоздания = СоздатьУниверсальныйЧерновикKaspi(Группа);
		КонецЕсли;

		Если РезультатСоздания.Создан Тогда
			Создано = Создано + 1;
			Если ПерваяСсылка = Неопределено Тогда
				ПерваяСсылка = РезультатСоздания.Ссылка;
			КонецЕсли;
		ИначеЕсли РезультатСоздания.Пропущен Тогда
			Если СтрНайти(РезультатСоздания.Текст,
					"ВНУТРЕННИЙ ПЕРЕВОД УЖЕ УЧТЕН") > 0
				Или СтрНайти(РезультатСоздания.Текст,
					"ЧЕРНОВИК ВНУТРЕННЕГО ПЕРЕВОДА УЖЕ ЕСТЬ") > 0

					"ДУБЛЬ ВНУТРЕННЕГО ПЕРЕВОДА") > 0 Тогда

			ИначеЕсли СтрНайти(РезультатСоздания.Текст, "ДУБЛЬ") > 0 Тогда
				ПропущеноДублей = ПропущеноДублей + 1;
			Иначе
				ПропущеноБезОбразца = ПропущеноБезОбразца + 1;
			КонецЕсли;
		Иначе
			Ошибок = Ошибок + 1;
		КонецЕсли;

		Сообщения.Добавить(РезультатСоздания.Текст);

		+ "; создано непроведенных: " + Строка(Создано)
		+ "; уже есть в 1С: " + Строка(ПропущеноДублей)
		+ "; внутренних переводов учтено другим банком: "
		+ Строка(УчтеноДругимБанком)
		+ "; на ручную проверку: "
		+ Строка(ПропущеноБезОбразца)
		+ "; ошибок: " + Строка(Ошибок) + ".";

	Если Сообщения.Количество() > 0 Тогда
		ТекстРезультата = ТекстРезультата + Символы.ПС
			+ СтрСоединить(Сообщения, Символы.ПС);
	КонецЕсли;

	Возврат Новый Структура(
		"Создан,Количество,Текст,Ссылка",
		Создано > 0, Создано, ТекстРезультата, ПерваяСсылка);

Функция СоздатьУниверсальныйЧерновикKaspi(Группа)

	МеткиРучнойПроверки = "";

	// Входящий собственный перевод в Kaspi не должен создавать вторую сторону.
	// Ищем исходящий документ банка-источника (например Alatau -> Kaspi).
	// Если он уже есть — ничего не создаем. Если его нет — создаем именно
	// НЕПРОВЕДЕННОЕ исходящее поручение со счетами, поменянными местами.
	Если Группа.Правило = "OWN_TRANSFER_IN" Тогда

		ПоискПеревода = НайтиПроведенныеИсходящиеПереводыKaspi(
			Группа.Дата, Группа.Сумма,
			Группа.СчетКлиента, Группа.ИИККорреспондента);

		Если ПоискПеревода.ПроверкаПолная Тогда

			Если ПоискПеревода.Проведенные.Количество() > 0 Тогда
				Возврат Новый Структура("Создан,Пропущен,Текст,Ссылка",
					Ложь, Истина,
					"ВНУТРЕННИЙ ПЕРЕВОД УЖЕ УЧТЕН ИСХОДЯЩИМ ДОКУМЕНТОМ: "
						+ Формат(Группа.Дата, "ДФ=dd.MM.yyyy") + " | "
						+ Формат(Группа.Сумма, "ЧДЦ=2") + " KZT | "
						+ СтрСоединить(ПоискПеревода.Проведенные, "; ")
						+ ". Входящий документ Kaspi не создан.",
					Неопределено);
			КонецЕсли;

			Если ПоискПеревода.Непроведенные.Количество() > 0 Тогда
				Возврат Новый Структура("Создан,Пропущен,Текст,Ссылка",
					Ложь, Истина,
					"ЧЕРНОВИК ВНУТРЕННЕГО ПЕРЕВОДА УЖЕ ЕСТЬ: "
						+ Формат(Группа.Дата, "ДФ=dd.MM.yyyy") + " | "
						+ Формат(Группа.Сумма, "ЧДЦ=2") + " KZT | "
						+ СтрСоединить(ПоискПеревода.Непроведенные, "; ")
						+ ". Второй документ не создан.",
					Неопределено);
			КонецЕсли;

		КонецЕсли;

		// Строим исходящую сторону из входящей строки Kaspi.
		// Важно: номер по банку входящей строки Kaspi не является номером
		// исходящего документа другого банка, поэтому его не переносим.
		ГруппаИсходящая = Новый Структура;
		ГруппаИсходящая.Вставить("Дата", Группа.Дата);
		ГруппаИсходящая.Вставить("Направление", "Списание");
		ГруппаИсходящая.Вставить("Сумма", Группа.Сумма);
		ГруппаИсходящая.Вставить("Корреспондент", Группа.Корреспондент);
		ГруппаИсходящая.Вставить("БИНКорреспондента", Группа.БИНКорреспондента);
		ГруппаИсходящая.Вставить("КНП", "343");
		ГруппаИсходящая.Вставить("Правило", "OWN_TRANSFER_OUT");
		ГруппаИсходящая.Вставить("НомерДокументаБанка", "");
		ГруппаИсходящая.Вставить("НазначениеПлатежа", Группа.НазначениеПлатежа);

		// Было: другой банк -> Kaspi.
		// Для исходящего документа источник и получатель меняются местами:
		// источник = ИИК корреспондента (другой собственный банк),
		// получатель = счет Kaspi из заголовка выписки.
		ГруппаИсходящая.Вставить("СчетКлиента", Группа.ИИККорреспондента);
		ГруппаИсходящая.Вставить("ИИККорреспондента", Группа.СчетКлиента);
		ГруппаИсходящая.Вставить("БИККорреспондента", "");

		Возврат СоздатьУниверсальныйЧерновикKaspi(ГруппаИсходящая);

	КонецЕсли;

	// Неизвестное правило тоже не блокирует явно выбранный пользователем
	// НЕПРОВЕДЕННЫЙ черновик; оно лишь требует ручной проверки.
	Если Группа.Правило <> "SALE"
		И Группа.Правило <> "ACCOUNTABLE_AURIKA"
		И Группа.Правило <> "OWN_TRANSFER_IN"
		И Группа.Правило <> "OWN_TRANSFER_OUT" Тогда
		МеткиРучнойПроверки = МеткиРучнойПроверки
			+ "[ РУЧНАЯ ПРОВЕРКА ] ";
	КонецЕсли;

	Если Группа.Направление = "Поступление" Тогда
		Если Группа.Правило = "SALE"
			Или Группа.Правило = "OWN_TRANSFER_IN" Тогда
			ИмяДокумента = "ПлатежноеПоручениеВходящее";
		Иначе
			ИмяДокумента = "ПлатежныйОрдерПоступлениеДенежныхСредств";
		КонецЕсли;
	Иначе
		Если Группа.Правило = "ACCOUNTABLE_AURIKA"
			Или Группа.Правило = "OWN_TRANSFER_OUT" Тогда
			ИмяДокумента = "ПлатежноеПоручениеИсходящее";
		Иначе
			ИмяДокумента = "ПлатежныйОрдерСписаниеДенежныхСредств";
		КонецЕсли;
	КонецЕсли;

	МетаданныеДокумента = Метаданные.Документы.Найти(ИмяДокумента);
	Если МетаданныеДокумента = Неопределено Тогда
		Возврат Новый Структура("Создан,Пропущен,Текст,Ссылка",
			Ложь, Ложь,
			"ОШИБКА: в конфигурации нет документа «" + ИмяДокумента + "».",
			Неопределено);
	КонецЕсли;

	ИмяРеквизитаКонтрагента = НайтиРеквизитКонтрагентаKaspi(
		МетаданныеДокумента, Группа.Направление);
	Контрагент = НайтиЕдинственногоКонтрагентаKaspiПоБИН(
		Группа.БИНКорреспондента);
	Образец = Неопределено;
	ОписаниеОбразца = "";

	КонецЕсли;

		КонецЕсли;

			ОбъектКандидата = Кандидат.Ссылка.ПолучитьОбъект();

		КонецПопытки;
	КонецЦикла;

		Попытка
			ОбъектСчета = СсылкаСчета.ПолучитьОбъект();

				Если ЗначениеЗаполнено(Организация) Тогда

			КонецПопытки;

Процедура УстановитьПодотчетникаВСтрокахKaspi(
		ДокументОбъект, МетаданныеДокумента, Подотчетник)

	Если Подотчетник = Неопределено Тогда
		Возврат;
	КонецЕсли;

	Для Каждого МетаТабличнаяЧасть Из МетаданныеДокумента.ТабличныеЧасти Цикл
		ИменаПолей = Новый Массив;
		Для Каждого МетаРеквизит Из МетаТабличнаяЧасть.Реквизиты Цикл
			ИмяНРег = НРег(МетаРеквизит.Имя);
			СинонимНРег = НРег(СокрЛП(Строка(МетаРеквизит.Синоним)));
			ТекстПоля = ИмяНРег + " " + СинонимНРег;
			Если СтрНайти(ТекстПоля, "подотчет") > 0
				И Не (СтрНайти(ТекстПоля, "вид") > 0
					И СтрНайти(ТекстПоля, "задолж") > 0) Тогда
				ИменаПолей.Добавить(МетаРеквизит.Имя);
			КонецЕсли;
		КонецЦикла;

		Если ИменаПолей.Количество() = 0 Тогда
			Продолжить;
		КонецЕсли;

		Попытка
			ТабличнаяЧасть = ДокументОбъект[МетаТабличнаяЧасть.Имя];
			Если ТабличнаяЧасть.Количество() = 0 Тогда
				ТабличнаяЧасть.Добавить();
			КонецЕсли;
			Для Каждого СтрокаТаблицы Из ТабличнаяЧасть Цикл
				Для Каждого ИмяПоля Из ИменаПолей Цикл
					Попытка
						СтрокаТаблицы[ИмяПоля] = Подотчетник;
					Исключение
					КонецПопытки;
				КонецЦикла;
			КонецЦикла;
		Исключение
		КонецПопытки;
	КонецЦикла;

Процедура ВосстановитьПоляПодотчетаИзОбразцаKaspi(

			Продолжить;

		СтрокаПриемника = Приемник[0];

					И СтрНайти(ТекстПоля, "задолж") > 0);

				И СтрНайти(ТекстПоля, "задолж") > 0;
			Если Не ЭтоПодотчетник И Не ЭтоВидЗадолженности Тогда
				Продолжить;
			КонецЕсли;

			Попытка
				ЗначениеОбразца = СтрокаИсточника[МетаРеквизит.Имя];
				Если ЗначениеЗаполнено(ЗначениеОбразца) Тогда
					СтрокаПриемника[МетаРеквизит.Имя] = ЗначениеОбразца;
				КонецЕсли;
			Исключение
			КонецПопытки;
		КонецЦикла;
	КонецЦикла;

Функция НайтиБанковскийСчетОрганизацииKaspi(НомерСчета)

	// 1.4.9: ищем IBAN/ИИК не только точным строковым сравнением.
	// В разных базах номер может храниться с пробелами, неразрывными
	// пробелами, дефисами или прямо в наименовании банковского счета.
	НомерСчетаНорм = НормализоватьНомерСчетаKaspi(НомерСчета);
	Если ПустаяСтрока(НомерСчетаНорм) Тогда
		Возврат Неопределено;
	КонецЕсли;

	Совпадения = Новый Массив;

	ИменаСправочников = Новый Массив;
	ИменаСправочников.Добавить("БанковскиеСчета");
	ИменаСправочников.Добавить("БанковскиеСчетаОрганизаций");

	Для Каждого ИмяСправочника Из ИменаСправочников Цикл

		МетаданныеСправочника = Метаданные.Справочники.Найти(ИмяСправочника);
		Если МетаданныеСправочника = Неопределено Тогда
			Продолжить;
		КонецЕсли;

		ИменаРеквизитов = Новый Массив;
		ЯвныеИмена = Новый Массив;
		ЯвныеИмена.Добавить("НомерСчета");
		ЯвныеИмена.Добавить("НомерСчетаБанка");
		ЯвныеИмена.Добавить("IBAN");
		ЯвныеИмена.Добавить("ИИК");

		Для Каждого ИмяРеквизита Из ЯвныеИмена Цикл
			Если МетаданныеСправочника.Реквизиты.Найти(ИмяРеквизита)
				<> Неопределено Тогда
				ИменаРеквизитов.Добавить(ИмяРеквизита);
			КонецЕсли;
		КонецЦикла;

		Для Каждого МетаРеквизит Из МетаданныеСправочника.Реквизиты Цикл
			ИмяНРег = НРег(МетаРеквизит.Имя);
			СинонимНРег = НРег(СокрЛП(Строка(МетаРеквизит.Синоним)));
			Подходит = СтрНайти(ИмяНРег, "iban") > 0
				Или СтрНайти(ИмяНРег, "иик") > 0
				Или СтрНайти(СинонимНРег, "iban") > 0
				Или СтрНайти(СинонимНРег, "иик") > 0
				Или ((СтрНайти(ИмяНРег, "номер") > 0
					И СтрНайти(ИмяНРег, "счет") > 0)
					Или (СтрНайти(СинонимНРег, "номер") > 0
						И СтрНайти(СинонимНРег, "счет") > 0));

			Если Подходит
				И ИменаРеквизитов.Найти(МетаРеквизит.Имя) = Неопределено Тогда
				ИменаРеквизитов.Добавить(МетаРеквизит.Имя);
			КонецЕсли;
		КонецЦикла;

		// Сначала пробуем быстрый точный поиск по реальным полям счета.
		Для Каждого ИмяРеквизита Из ИменаРеквизитов Цикл
			Запрос = Новый Запрос(
				"ВЫБРАТЬ Счета.Ссылка КАК Ссылка "
				+ "ИЗ Справочник." + ИмяСправочника + " КАК Счета "
				+ "ГДЕ НЕ Счета.ПометкаУдаления "
				+ "И Счета." + ИмяРеквизита + " = &НомерСчета");
			Запрос.УстановитьПараметр("НомерСчета", СокрЛП(Строка(НомерСчета)));

			Попытка
				Выборка = Запрос.Выполнить().Выбрать();
				Пока Выборка.Следующий() Цикл
					Если Совпадения.Найти(Выборка.Ссылка) = Неопределено Тогда
						Совпадения.Добавить(Выборка.Ссылка);
					КонецЕсли;
				КонецЦикла;
			Исключение
			КонецПопытки;
		КонецЦикла;

		// Если точное сравнение не сработало, читаем номера и сравниваем
		// после нормализации. Это ловит пробелы, дефисы и NBSP.
		Если Совпадения.Количество() = 0 Тогда
			Для Каждого ИмяРеквизита Из ИменаРеквизитов Цикл
				Запрос = Новый Запрос(
					"ВЫБРАТЬ Счета.Ссылка КАК Ссылка, "
					+ "Счета." + ИмяРеквизита + " КАК ЗначениеСчета "
					+ "ИЗ Справочник." + ИмяСправочника + " КАК Счета "
					+ "ГДЕ НЕ Счета.ПометкаУдаления");

				Попытка
					Выборка = Запрос.Выполнить().Выбрать();
					Пока Выборка.Следующий() Цикл
						КандидатНорм = НормализоватьНомерСчетаKaspi(Выборка.ЗначениеСчета);
						Если КандидатНорм = НомерСчетаНорм Тогда
							Если Совпадения.Найти(Выборка.Ссылка) = Неопределено Тогда
								Совпадения.Добавить(Выборка.Ссылка);
							КонецЕсли;
						КонецЕсли;
					КонецЦикла;
				Исключение
				КонецПопытки;
			КонецЦикла;
		КонецЕсли;

		// Последний безопасный вариант: в некоторых базах IBAN включен
		// непосредственно в Наименование элемента банковского счета.
		Если Совпадения.Количество() = 0
			И СтрДлина(НомерСчетаНорм) >= 10 Тогда

			Запрос = Новый Запрос(
				"ВЫБРАТЬ Счета.Ссылка КАК Ссылка, "
				+ "Счета.Наименование КАК Наименование "
				+ "ИЗ Справочник." + ИмяСправочника + " КАК Счета "
				+ "ГДЕ НЕ Счета.ПометкаУдаления");

			Попытка
				Выборка = Запрос.Выполнить().Выбрать();
				Пока Выборка.Следующий() Цикл
					НаименованиеНорм = НормализоватьНомерСчетаKaspi(Выборка.Наименование);
					Если СтрНайти(НаименованиеНорм, НомерСчетаНорм) > 0 Тогда
						Если Совпадения.Найти(Выборка.Ссылка) = Неопределено Тогда
							Совпадения.Добавить(Выборка.Ссылка);
						КонецЕсли;
					КонецЕсли;
				КонецЦикла;
			Исключение
			КонецПопытки;
		КонецЕсли;

	КонецЦикла;

	Если Совпадения.Количество() = 1 Тогда
		Возврат Совпадения[0];
	КонецЕсли;

	// Если в базе есть дубли одного IBAN, но все они принадлежат
	// одной и той же организации, это не считаем опасной неоднозначностью.
	Если Совпадения.Количество() > 1 Тогда
		ОбщийВладелец = Неопределено;
		ОдинаковыйВладелец = Истина;

		Для Каждого СсылкаСчета Из Совпадения Цикл
			ВладелецСчета = ПолучитьОрганизациюБанковскогоСчетаKaspi(СсылкаСчета);
			Если ВладелецСчета = Неопределено Тогда
				ОдинаковыйВладелец = Ложь;
				Прервать;
			КонецЕсли;

			Если ОбщийВладелец = Неопределено Тогда
				ОбщийВладелец = ВладелецСчета;
			ИначеЕсли ВладелецСчета <> ОбщийВладелец Тогда
				ОдинаковыйВладелец = Ложь;
				Прервать;
			КонецЕсли;
		КонецЦикла;

		Если ОдинаковыйВладелец И ОбщийВладелец <> Неопределено Тогда
			Возврат Совпадения[0];
		КонецЕсли;
	КонецЕсли;

	Возврат Неопределено;

Функция НайтиЕдинственногоКонтрагентаKaspiПоБИН(БИН)

	БИН = СокрЛП(Строка(БИН));
	Если ПустаяСтрока(БИН) Тогда
		Возврат Неопределено;
	КонецЕсли;

	// Нулевой тестовый/служебный идентификатор не является настоящим БИН.
	// В некоторых базах у элемента «Розничная выручка» может быть записано
	// такое значение, поэтому искать контрагента по нему небезопасно.
	Если СтрЗаменить(БИН, "0", "") = "" Тогда
		Возврат Неопределено;
	КонецЕсли;

	МетаданныеКонтрагентов = Метаданные.Справочники.Найти("Контрагенты");
	Если МетаданныеКонтрагентов = Неопределено Тогда
		Возврат Неопределено;
	КонецЕсли;

	// В разных релизах Бухгалтерии для Казахстана поле БИН/ИИН
	// имеет разные внутренние имена. Сначала добавляем известные варианты.
	Варианты = Новый Массив;
	ЯвныеИмена = Новый Массив;
	ЯвныеИмена.Добавить("БИН");
	ЯвныеИмена.Добавить("ИИН");
	ЯвныеИмена.Добавить("БИН_ИИН");
	ЯвныеИмена.Добавить("БИНИИН");
	ЯвныеИмена.Добавить("БИНФизЛица");
	ЯвныеИмена.Добавить("ИдентификационныйНомер");
	ЯвныеИмена.Добавить("ИдентификационныйНомерНалогоплательщика");

	Для Каждого ИмяРеквизита Из ЯвныеИмена Цикл
		Если МетаданныеКонтрагентов.Реквизиты.Найти(ИмяРеквизита) <> Неопределено Тогда
			Варианты.Добавить(ИмяРеквизита);
		КонецЕсли;
	КонецЦикла;

	// Дополнительно определяем поле по имени и синониму метаданных.
	// Это позволяет работать с релизами, где внутреннее имя заранее неизвестно.
	Для Каждого МетаРеквизит Из МетаданныеКонтрагентов.Реквизиты Цикл
		ИмяНРег = НРег(МетаРеквизит.Имя);
		СинонимНРег = НРег(СокрЛП(Строка(МетаРеквизит.Синоним)));
		ЭтоБИН = СтрНайти(ИмяНРег, "бин") > 0
			Или СтрНайти(ИмяНРег, "иин") > 0
			Или СтрНайти(ИмяНРег, "идентификацион") > 0
			Или СтрНайти(СинонимНРег, "бин") > 0
			Или СтрНайти(СинонимНРег, "иин") > 0
			Или СтрНайти(СинонимНРег, "идентификацион") > 0;

		Если ЭтоБИН И Варианты.Найти(МетаРеквизит.Имя) = Неопределено Тогда
			Варианты.Добавить(МетаРеквизит.Имя);

Функция ТекстСодержитТочныйКНПKaspi(ТекстКНП, ИскомыйКНП)

	ТекстКНП = НормализоватьТекст(ТекстКНП);
	ИскомыйКНП = СокрЛП(Строка(ИскомыйКНП));
	Если ПустаяСтрока(ТекстКНП) Или ПустаяСтрока(ИскомыйКНП) Тогда
		Возврат Ложь;
	КонецЕсли;

	ПоследняяПозиция = СтрДлина(ТекстКНП) - СтрДлина(ИскомыйКНП) + 1;
	Если ПоследняяПозиция < 1 Тогда
		Возврат Ложь;
	КонецЕсли;
	Для Позиция = 1 По ПоследняяПозиция Цикл
		Если Сред(ТекстКНП, Позиция, СтрДлина(ИскомыйКНП))
			<> ИскомыйКНП Тогда
			Продолжить;
		КонецЕсли;
		ПредыдущийСимвол = "";
		СледующийСимвол = "";
		Если Позиция > 1 Тогда
			ПредыдущийСимвол = Сред(ТекстКНП, Позиция - 1, 1);
		КонецЕсли;
		Если Позиция + СтрДлина(ИскомыйКНП) <= СтрДлина(ТекстКНП) Тогда
			СледующийСимвол = Сред(
				ТекстКНП, Позиция + СтрДлина(ИскомыйКНП), 1);
		КонецЕсли;
		Если (ПустаяСтрока(ПредыдущийСимвол)
				Или СтрНайти("0123456789", ПредыдущийСимвол) = 0)
			И (ПустаяСтрока(СледующийСимвол)
				Или СтрНайти("0123456789", СледующийСимвол) = 0) Тогда
			Возврат Истина;
		КонецЕсли;
	КонецЦикла;

	Возврат Ложь;

Функция ДокументСодержитБанковскийСчетKaspi(
		ДокументОбъект, МетаданныеДокумента,
		СсылкаСчета, НомерСчета)

		Если СтрНайти(ИмяНРег, "счет") = 0
			И СтрНайти(СинонимНРег, "счет") = 0 Тогда
			Продолжить;
		КонецЕсли;

		Попытка
			ЗначениеСчета = ДокументОбъект[МетаРеквизит.Имя];
		Исключение
			Продолжить;
		КонецПопытки;

		Если СсылкаСчета <> Неопределено Тогда
			Попытка
				Если ЗначениеСчета = СсылкаСчета Тогда
					Возврат Истина;
				КонецЕсли;
			Исключение
			КонецПопытки;
		КонецЕсли;

		Если Не ПустаяСтрока(НомерСчетаНорм) Тогда
			ТекстЗначения = НормализоватьНомерСчетаKaspi(
				Строка(ЗначениеСчета));
			Если СтрНайти(ТекстЗначения, НомерСчетаНорм) > 0 Тогда
				Возврат Истина;
			КонецЕсли;

			Попытка
				ОбъектСчета = ЗначениеСчета.ПолучитьОбъект();
				ИменаНомераСчета = Новый Массив;
				ИменаНомераСчета.Добавить("НомерСчета");
				ИменаНомераСчета.Добавить("НомерСчетаБанка");
				ИменаНомераСчета.Добавить("IBAN");
				ИменаНомераСчета.Добавить("ИИК");
				Для Каждого ИмяНомера Из ИменаНомераСчета Цикл
					Попытка
						ТекстНомера = НормализоватьНомерСчетаKaspi(
							ОбъектСчета[ИмяНомера]);
						Если ТекстНомера = НомерСчетаНорм Тогда
							Возврат Истина;
						КонецЕсли;
					Исключение
					КонецПопытки;
				КонецЦикла;
			Исключение
			КонецПопытки;
		КонецЕсли;
	КонецЦикла;

	Возврат Ложь;

Функция УстановитьВидОперацииУниверсальныйKaspi(
		ДокументОбъект, МетаданныеДокумента,
		Направление, Правило, Контрагент)

	МетаРеквизит = МетаданныеДокумента.Реквизиты.Найти("ВидОперации");
	Если МетаРеквизит = Неопределено Тогда
		Возврат Истина;
	КонецЕсли;

	ИмяПеречисления = НайтиИмяПеречисленияВидаОперацииKaspi(МетаРеквизит);
	Если ПустаяСтрока(ИмяПеречисления) Тогда
		Возврат Ложь;
	КонецЕсли;

	МетаПеречисление = Метаданные.Перечисления.Найти(ИмяПеречисления);
	Если МетаПеречисление = Неопределено Тогда
		Возврат Ложь;
	КонецЕсли;

	Кандидаты = Новый Массив;
	Если Направление = "Поступление" Тогда
		Если Правило = "SALE" Тогда
			Кандидаты.Добавить("ОплатаОтПокупателя");
			Кандидаты.Добавить("ПоступлениеОплатыОтПокупателя");
		ИначеЕсли Правило = "OWN_TRANSFER_IN" Тогда
			Кандидаты.Добавить("ПереводМеждуСчетамиОрганизации");
			Кандидаты.Добавить("ПереводМеждуБанковскимиСчетами");
			Кандидаты.Добавить("ПереводСДругогоСчетаОрганизации");
			Кандидаты.Добавить("ПоступлениеСДругогоСчетаОрганизации");
		КонецЕсли;
		Кандидаты.Добавить("ПрочееПоступлениеБезналичныхДенежныхСредств");
		Кандидаты.Добавить("ПрочееПоступление");
	Иначе
		Если Правило = "ACCOUNTABLE_AURIKA" Тогда
			Кандидаты.Добавить("ПеречислениеДенежныхСредствПодотчетнику");
			Кандидаты.Добавить("ПеречислениеДенежныхСредствПодотчетномуЛицу");
			Кандидаты.Добавить("ПеречислениеПодотчетнику");
			Кандидаты.Добавить("ВыдачаДенежныхСредствПодотчетнику");
		ИначеЕсли Правило = "OWN_TRANSFER_OUT" Тогда
			Кандидаты.Добавить("ПереводНаДругойСчетОрганизации");
			Кандидаты.Добавить("ПереводМеждуСчетамиОрганизации");
			Кандидаты.Добавить("ПереводМеждуБанковскимиСчетами");
		ИначеЕсли Контрагент <> Неопределено
			И Правило <> "MANUAL_REVIEW" Тогда
			Кандидаты.Добавить("ОплатаПоставщику");
		КонецЕсли;
		Если Правило <> "ACCOUNTABLE_AURIKA"
			И Правило <> "OWN_TRANSFER_OUT" Тогда
			Кандидаты.Добавить("ПрочееСписаниеБезналичныхДенежныхСредств");
			Кандидаты.Добавить("ПрочееСписание");
			Кандидаты.Добавить("ОплатаПоставщику");
		КонецЕсли;
	КонецЕсли;

	Для Каждого ИмяКандидата Из Кандидаты Цикл
		Для Каждого МетаЗначение Из МетаПеречисление.ЗначенияПеречисления Цикл
			Если МетаЗначение.Имя <> ИмяКандидата Тогда
				Продолжить;
			КонецЕсли;
			Если УстановитьЗначениеВидаОперацииKaspi(
					ДокументОбъект, ИмяПеречисления, МетаЗначение.Имя) Тогда
				Возврат Истина;
			КонецЕсли;
		КонецЦикла;
	КонецЦикла;

	// Запасной поиск по синонимам для разных релизов БК 3.0.
	Для Каждого МетаЗначение Из МетаПеречисление.ЗначенияПеречисления Цикл
		ИмяНРег = НРег(МетаЗначение.Имя);
		СинонимНРег = НРег(СокрЛП(Строка(МетаЗначение.Синоним)));
		ТекстПоиска = ИмяНРег + " " + СинонимНРег;

		Если Направление = "Поступление" Тогда
			Если Правило = "SALE" Тогда
				Подходит = СтрНайти(ТекстПоиска, "оплат") > 0
					И СтрНайти(ТекстПоиска, "покупател") > 0;
			ИначеЕсли Правило = "OWN_TRANSFER_IN" Тогда
				Подходит = СтрНайти(ТекстПоиска, "перевод") > 0
					И СтрНайти(ТекстПоиска, "счет") > 0;
			Иначе
				Подходит = СтрНайти(ТекстПоиска, "проч") > 0
					И СтрНайти(ТекстПоиска, "поступ") > 0;
			КонецЕсли;
		Иначе
			Если Правило = "ACCOUNTABLE_AURIKA" Тогда
				Подходит = СтрНайти(ТекстПоиска, "подотч") > 0
					И (СтрНайти(ТекстПоиска, "перечис") > 0
						Или СтрНайти(ТекстПоиска, "выдач") > 0);
			ИначеЕсли Правило = "OWN_TRANSFER_OUT" Тогда
				Подходит = СтрНайти(ТекстПоиска, "перевод") > 0
					И СтрНайти(ТекстПоиска, "друг") > 0
					И СтрНайти(ТекстПоиска, "счет") > 0
					И СтрНайти(ТекстПоиска, "организац") > 0;
			Иначе
				Подходит = СтрНайти(ТекстПоиска, "проч") > 0
					И СтрНайти(ТекстПоиска, "списан") > 0;
			КонецЕсли;
		КонецЕсли;

		Если Не Подходит Тогда
			Продолжить;
		КонецЕсли;
		Если УстановитьЗначениеВидаОперацииKaspi(
				ДокументОбъект, ИмяПеречисления, МетаЗначение.Имя) Тогда
			Возврат Истина;
		КонецЕсли;
	КонецЦикла;

	Возврат Ложь;

			И МетаТабличнаяЧасть.Реквизиты.Найти("Сумма") = Неопределено Тогда
			Продолжить;
		КонецЕсли;

			ДокументОбъект, МетаТабличнаяЧасть, СуммаДокумента);

		Если МетаТабличнаяЧасть.Реквизиты.Найти("СуммаПлатежа") = Неопределено Тогда

		УстановитьРеквизитKaspiЕслиЕсть(
			ДокументОбъект, МетаданныеДокумента,
			ИмяРеквизита, ДатаБанка);
	КонецЦикла;

	ИменаДатДокумента = Новый Массив;
	ИменаДатДокумента.Добавить("ДатаПлатежногоДокумента");
	ИменаДатДокумента.Добавить("ДатаПлатежногоПоручения");
	ИменаДатДокумента.Добавить("ДатаПлатежа");
	ИменаДатДокумента.Добавить("ДатаОплаты");
	ИменаДатДокумента.Добавить("ДатаПоручения");
	ИменаДатДокумента.Добавить("ДатаДокумента");
	ИменаДатДокумента.Добавить("ДатаВыписки");
	Для Каждого ИмяРеквизита Из ИменаДатДокумента Цикл
		УстановитьРеквизитKaspiЕслиЕсть(
			ДокументОбъект, МетаданныеДокумента,
			ИмяРеквизита, ДатаДокумента);
	КонецЦикла;

	// В этой базе верхнее поле «от:» платежного поручения хранится
	// в реквизите ДатаВыписки, а не только в системном Документ.Дата.
	// Поэтому закрепляем его отдельно после копирования образца.
	УстановитьРеквизитKaspiЕслиЕсть(
		ДокументОбъект, МетаданныеДокумента,
		"ДатаВыписки", ДатаДокумента);

	// Дата валютирования относится к текущей банковской операции.
	УстановитьРеквизитKaspiЕслиЕсть(
		ДокументОбъект, МетаданныеДокумента,
		"ДатаВалютирования", ДатаБанка);

	// В разных релизах БК 3.0 внутренние имена отличаются. Поэтому
	// дополнительно определяем поля по имени и синониму метаданных.
	Для Каждого МетаРеквизит Из МетаданныеДокумента.Реквизиты Цикл
		Если ЭтоРеквизитНомераПоБанкуKaspi(МетаРеквизит) Тогда
			УстановитьРеквизитKaspiЕслиЕсть(
				ДокументОбъект, МетаданныеДокумента,
				МетаРеквизит.Имя, НомерДокументаБанка);
		ИначеЕсли ЭтоРеквизитДатыПоБанкуKaspi(МетаРеквизит) Тогда
			ИмяНРег = НРег(МетаРеквизит.Имя);
			СинонимНРег = НРег(СокрЛП(Строка(МетаРеквизит.Синоним)));
			ТекстПоля = ИмяНРег + " " + СинонимНРег;
			Если СтрНайти(ТекстПоля, "банк") > 0
				И СтрНайти(ТекстПоля, "платеж") = 0
				И СтрНайти(ТекстПоля, "оплат") = 0
				И СтрНайти(ТекстПоля, "поруч") = 0 Тогда
				ЗначениеДаты = ДатаБанка;
			Иначе
				ЗначениеДаты = ДатаДокумента;
			КонецЕсли;
			УстановитьРеквизитKaspiЕслиЕсть(
				ДокументОбъект, МетаданныеДокумента,
				МетаРеквизит.Имя, ЗначениеДаты);
		КонецЕсли;
	КонецЦикла;

	УстановитьОсновнуюДатуЧерновикаKaspi(
		ДокументОбъект, ДатаДокумента, Истина);

Функция ОшибкиКритичныхРеквизитовЧерновикаKaspi(
		ДокументОбъект, МетаданныеДокумента, Группа, ДатаДокумента)

	Ошибки = Новый Массив;
	ОсновнаяДатаВБазе = ПолучитьОсновнуюДатуЧерновикаИзБазыKaspi(
		ДокументОбъект.Ссылка, МетаданныеДокумента);
	Если ОсновнаяДатаВБазе = Неопределено Тогда
		Ошибки.Добавить("не удалось прочитать основную дату документа из базы");
	ИначеЕсли ОсновнаяДатаВБазе <> ДатаДокумента Тогда
		Ошибки.Добавить("дата документа не равна "
			+ Формат(ДатаДокумента, "ДФ=dd.MM.yyyy HH:mm:ss")
			+ "; в базе "
			+ Формат(ОсновнаяДатаВБазе, "ДФ=dd.MM.yyyy HH:mm:ss"));
	КонецЕсли;

	Если МетаданныеДокумента.Реквизиты.Найти("ДатаВыписки") <> Неопределено Тогда
		Попытка
			ФактическаяДатаВыписки = ДокументОбъект.ДатаВыписки;
			Если Не ЗначениеЗаполнено(ФактическаяДатаВыписки)
				Или ФактическаяДатаВыписки <> ДатаДокумента Тогда
				Ошибки.Добавить("не заполнена верхняя дата «от:» (ДатаВыписки)");
			КонецЕсли;
		Исключение
			Ошибки.Добавить("не удалось проверить верхнюю дату «от:» (ДатаВыписки)");
		КонецПопытки;
	КонецЕсли;

	ИмяРеквизитаСуммы = НайтиРеквизитСуммы(МетаданныеДокумента);
	Если ПустаяСтрока(ИмяРеквизитаСуммы) Тогда
		Ошибки.Добавить("не найден реквизит суммы документа");
	Иначе
		Попытка
			Если ДокументОбъект[ИмяРеквизитаСуммы] <> Группа.Сумма Тогда
				Ошибки.Добавить("сумма документа не равна сумме выписки");
			КонецЕсли;
		Исключение
			Ошибки.Добавить("не удалось проверить сумму документа");
		КонецПопытки;
	КонецЕсли;

	Если Группа.Правило = "OWN_TRANSFER_OUT" Тогда
		// Для НЕПРОВЕДЕННОГО черновика отсутствие/неоднозначность банковского
		// счета больше не является фатальной ошибкой: это помечается в комментарии
		// и исправляется бухгалтером перед проведением.
		Если Не ДокументИмеетКНПKaspi(
				ДокументОбъект, МетаданныеДокумента, "343") Тогда
			Ошибки.Добавить("не установлен точный КНП 343");
		КонецЕсли;
		Если Не ЭтоВидОперацииВнутреннегоПереводаKaspi(
				ДокументОбъект, МетаданныеДокумента) Тогда
			Ошибки.Добавить("не установлен вид операции внутреннего перевода");
		КонецЕсли;
	КонецЕсли;

	ОжидаемыйНомер = СокрЛП(Строка(Группа.НомерДокументаБанка));
	Для Каждого МетаРеквизит Из МетаданныеДокумента.Реквизиты Цикл
		Если ЭтоРеквизитНомераПоБанкуKaspi(МетаРеквизит) Тогда
			Попытка
				ФактическийНомер = СокрЛП(Строка(
					ДокументОбъект[МетаРеквизит.Имя]));
				Если ФактическийНомер <> ОжидаемыйНомер Тогда
					Ошибки.Добавить("не обновлён номер по банку «"
						+ МетаРеквизит.Имя + "»");
				КонецЕсли;
			Исключение
			КонецПопытки;
		ИначеЕсли ЭтоРеквизитДатыПоБанкуKaspi(МетаРеквизит) Тогда
			Попытка
				ФактическаяДата = ДокументОбъект[МетаРеквизит.Имя];
				Если Не ЗначениеЗаполнено(ФактическаяДата)
					Или НачалоДня(ФактическаяДата) <> НачалоДня(Группа.Дата) Тогда
					Ошибки.Добавить("не обновлена банковская дата «"
						+ МетаРеквизит.Имя + "»");
				КонецЕсли;
			Исключение
			КонецПопытки;
		КонецЕсли;
	КонецЦикла;

	Если Группа.Правило = "ACCOUNTABLE_AURIKA"
		И Не ПодотчетникВСтрокахЗаполненKaspi(
			ДокументОбъект, МетаданныеДокумента) Тогда
		Ошибки.Добавить("в строке не заполнен подотчётник");
	КонецЕсли;
	Если Группа.Правило = "ACCOUNTABLE_AURIKA"
		И Не ВидЗадолженностиПодотчетаЗаполненKaspi(
			ДокументОбъект, МетаданныеДокумента) Тогда
		Ошибки.Добавить("в строке не заполнен вид задолженности подотчётного лица");
	КонецЕсли;

	Возврат СтрСоединить(Ошибки, "; ");

КонецФункции

Функция ПроверитьИИсправитьСохраненныйЧерновикKaspi(
		СсылкаДокумента, МетаданныеДокумента, Группа,
		Образец, ДатаДокумента)

	Попытка
		КонтрольныйОбъект = СсылкаДокумента.ПолучитьОбъект();
	Исключение
		Возврат "не удалось повторно прочитать сохранённый документ: "
			+ ОписаниеОшибки();
	КонецПопытки;

	Ошибки = ОшибкиКритичныхРеквизитовЧерновикаKaspi(
		КонтрольныйОбъект, МетаданныеДокумента, Группа, ДатаДокумента);
	Если ПустаяСтрока(Ошибки) Тогда
		Возврат "";
	КонецЕсли;

	КонтрольныйОбъект.ОбменДанными.Загрузка = Истина;
	Если Группа.Правило = "ACCOUNTABLE_AURIKA" Тогда
		ВосстановитьПоляПодотчетаИзОбразцаKaspi(
			КонтрольныйОбъект, Образец, МетаданныеДокумента);
	КонецЕсли;
	Если Группа.Правило = "OWN_TRANSFER_OUT" Тогда
		УстановитьВидОперацииУниверсальныйKaspi(
			КонтрольныйОбъект, МетаданныеДокумента,
			"Списание", Группа.Правило, Неопределено);
		УстановитьБанковскийСчетОрганизацииKaspi(
			КонтрольныйОбъект, МетаданныеДокумента,
			Группа.СчетКлиента);
		УстановитьБанковскийСчетПолучателяKaspi(
			КонтрольныйОбъект, МетаданныеДокумента,
			Группа.ИИККорреспондента);
		УстановитьРеквизитKaspiЕслиЕсть(
			КонтрольныйОбъект, МетаданныеДокумента,
			"КНП", Группа.КНП);
		УстановитьРеквизитKaspiЕслиЕсть(
			КонтрольныйОбъект, МетаданныеДокумента,
			"КодНазначенияПлатежа", Группа.КНП);
	КонецЕсли;
	УстановитьБанковскиеПоляKaspi(
		КонтрольныйОбъект, МетаданныеДокумента,
		Группа.НомерДокументаБанка, Группа.Дата, ДатаДокумента);
	УстановитьОсновнуюДатуЧерновикаKaspi(
		КонтрольныйОбъект, ДатаДокумента, Истина);

	Попытка
		КонтрольныйОбъект.Записать();
	Исключение
		Возврат "повторная запись исправленных реквизитов не выполнена: "
			+ ОписаниеОшибки();
	КонецПопытки;

	Попытка
		КонтрольныйОбъект = СсылкаДокумента.ПолучитьОбъект();
	Исключение
		Возврат "не удалось проверить документ после повторной записи: "
			+ ОписаниеОшибки();
	КонецПопытки;

	Возврат ОшибкиКритичныхРеквизитовЧерновикаKaspi(
		КонтрольныйОбъект, МетаданныеДокумента, Группа, ДатаДокумента);

Функция СформироватьИтоги(Строки, Группы)

	Дебет = 0;
	Кредит = 0;
	АвтоматическиеСтроки = 0;
	РучныеСтроки = 0;

	Для Каждого СтрокаВыписки Из Строки Цикл
		Дебет = Дебет + СтрокаВыписки.Дебет;
		Кредит = Кредит + СтрокаВыписки.Кредит;
		Если ЭтоПоддерживаемоеПравилоKaspi(СтрокаВыписки.Правило) Тогда
			АвтоматическиеСтроки = АвтоматическиеСтроки + 1;
		Иначе
			РучныеСтроки = РучныеСтроки + 1;
		КонецЕсли;
	КонецЦикла;

	Возврат
		"Строк: " + Строка(Строки.Количество()) + Символы.ПС
		+ "Групп: " + Строка(Группы.Количество()) + Символы.ПС
		+ "Дебет: " + Формат(Дебет, "ЧДЦ=2") + Символы.ПС
		+ "Кредит: " + Формат(Кредит, "ЧДЦ=2") + Символы.ПС
		+ "Чистое движение: " + Формат(Кредит - Дебет, "ЧДЦ=2") + Символы.ПС
		+ "Поддержано правилами: " + Строка(АвтоматическиеСтроки) + Символы.ПС
		+ "Ручная проверка: " + Строка(РучныеСтроки);
